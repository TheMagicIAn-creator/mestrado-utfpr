@st.cache_resource
def carregar_base():
    """
    Carrega embeddings e ChromaDB uma Ãºnica vez.
    Executa tambÃ©m o orquestrador de backend.
    O @st.cache_resource evita recarregar a cada interaÃ§Ã£o.
    """
    from sentence_transformers import SentenceTransformer
    import chromadb

    from src.core.config import (
        MODELO_EMBEDDINGS,
        PASTA_CHROMADB,
        NOME_COLECAO,
        NOME_COLECAO_SESSOES
    )

    with st.spinner("ðŸ”„ Carregando modelo de embeddings..."):
        modelo = SentenceTransformer(MODELO_EMBEDDINGS)

        # â”€â”€ Inicia watcher em background â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

            iniciar_em_background(modelo)
        except Exception:
            pass  # silencioso â€” nÃ£o bloqueia o Streamlit

    # â”€â”€ Orquestrador de backend â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    relatorio_orquestrador = []
    with st.spinner("âš™ï¸ Verificando estado do projeto..."):
        try:
            from src.orquestrador import executar_pipeline
            relatorio_orquestrador = executar_pipeline(modelo)
            # Registra o relatÃ³rio completo no terminal
            print("=" * 60)
            print("  ORQUESTRADOR â€” RELATÃ“RIO DE INICIALIZAÃ‡ÃƒO")
            print("=" * 60)
            for linha in relatorio_orquestrador:
                print(f"  {linha}")
            print("=" * 60)
        except Exception as e:
            print(f"[Orquestrador] Erro: {e}")

    with st.spinner("ðŸ—„ï¸ Conectando ao ChromaDB..."):
        client          = chromadb.PersistentClient(path=str(PASTA_CHROMADB))
        colecao         = client.get_or_create_collection(name=NOME_COLECAO)
        colecao_sessoes = client.get_or_create_collection(name=NOME_COLECAO_SESSOES)

    return perfil, modelo, colecao, colecao_sessoes, relatorio_orquestrador

def mostrar_novidades_orquestrador(relatorio: list):
    """
    Exibe aviso apenas quando o orquestrador processou algo novo.
    Mensagens de 'nada pendente' / 'jÃ¡ realizada' sÃ£o silenciadas.
    """
    # Palavras que indicam que NADA novo aconteceu
    termos_inertes = ["nenhum", "sem acÃºmulo", "jÃ¡ realizada"]

    novidades = [
        linha for linha in relatorio
        if not any(termo in linha for termo in termos_inertes)
    ]

    if novidades:
        with st.expander("âš™ï¸ O sistema processou novidades", expanded=True):
            for linha in novidades:
                if "erro" in linha.lower():
                    st.warning(f"âš ï¸ {linha}")
                else:
                    st.success(f"âœ… {linha}")

def stream_resposta(prompt: str, llm):
    """
    Generator que faz streaming do LLM para o Streamlit.
    CompatÃ­vel com st.write_stream().
    """
    mensagens = [HumanMessage(content=prompt)]
    for chunk in llm.stream(mensagens):
        yield chunk.content

def salvar_sessao_streamlit(pergunta: str, resposta: str, n: int, modelo_embeddings):
    """
    Cria o arquivo de sessÃ£o na primeira interaÃ§Ã£o
    e ADICIONA ao mesmo arquivo nas seguintes.
    """
    from datetime import datetime
    from src.core.config import PASTA_CHROMADB

    pasta_sessoes = Path(__file__).parent / "notas" / "sessoes"
    pasta_sessoes.mkdir(parents=True, exist_ok=True)

    # â”€â”€ Primeira interaÃ§Ã£o â€” cria o arquivo â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    if st.session_state.caminho_sessao is None:
        agora        = datetime.now()
        nome_arquivo = agora.strftime("%Y-%m-%d_%H-%M") + "_sessao_web.md"
        caminho      = pasta_sessoes / nome_arquivo
        data_fmt     = agora.strftime("%d/%m/%Y Ã s %H:%M")

        cabecalho  = f"---\n"
        cabecalho += f"data: {agora.strftime('%Y-%m-%d')}\n"
        cabecalho += f"hora: {agora.strftime('%H:%M')}\n"
        cabecalho += f"tipo: sessao-web\n"
        cabecalho += f"tags: [al-iado-pv, sessao, streamlit, mestrado]\n"
        cabecalho += f"---\n\n"
        cabecalho += f"# SessÃ£o Web Al IAdo PV â€” {data_fmt}\n\n"

        caminho.write_text(cabecalho, encoding="utf-8")
        st.session_state.caminho_sessao = caminho

    # â”€â”€ Todas as interaÃ§Ãµes â€” adiciona ao mesmo arquivo â”€â”€â”€â”€â”€
    caminho = st.session_state.caminho_sessao

    bloco  = f"---\n\n"
    bloco += f"## InteraÃ§Ã£o {n}\n\n"
    bloco += f"**ðŸ”¬ VocÃª:** {pergunta}\n\n"
    bloco += f"**ðŸ¤– Al IAdo PV:**\n\n{resposta}\n\n"

    with open(caminho, "a", encoding="utf-8") as f:
        f.write(bloco)

    # â”€â”€ Indexa no ChromaDB a cada interaÃ§Ã£o â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    if n >= 1:
        try:
            from src.conhecimento.indexador import indexar_sessao
            indexar_sessao(caminho, modelo_embeddings, PASTA_CHROMADB)
        except Exception:
            pass  # silencioso â€” nÃ£o interrompe o chat

def main():

    # Inicializa session_state
    if "mensagens"     not in st.session_state:
        st.session_state.mensagens      = []
    if "llm"           not in st.session_state:
        st.session_state.llm            = None
    if "nome_provedor" not in st.session_state:
        st.session_state.nome_provedor  = "Nenhum"
    if "mostrar_artigos" not in st.session_state:
        st.session_state.mostrar_artigos = False
    if "caminho_sessao" not in st.session_state:
        st.session_state.caminho_sessao = None

    # Carrega componentes base + orquestrador
    try:
        perfil, modelo, colecao, colecao_sessoes, relatorio = carregar_base()
    except Exception as e:
        st.error(f"âŒ Erro ao carregar o agente: {e}")
        st.info("Verifique a indexaÃ§Ã£o da base de conhecimento.")
        return

    # Sidebar
    renderizar_sidebar(perfil, modelo, colecao, colecao_sessoes)

    # TÃ­tulo principal
    st.title("âš¡ Al IAdo PV")
    st.caption("Assistente especialista em inversores fotovoltaicos â€” UTFPR")

    # Feedback do orquestrador â€” sÃ³ aparece se houve novidade
    mostrar_novidades_orquestrador(relatorio)

    # Listagem de artigos (se solicitada)
    if st.session_state.mostrar_artigos:
        from src.conhecimento.agente import listar_documentos
        with st.expander("ðŸ“š Artigos indexados", expanded=True):
            st.text(listar_documentos(colecao))
        st.session_state.mostrar_artigos = False

    # Verifica se o LLM estÃ¡ conectado
    if st.session_state.llm is None:
        st.info("ðŸ‘ˆ Selecione um provedor de LLM no painel lateral e clique em **Conectar provedor** para comeÃ§ar.")
        return

    # Exibe histÃ³rico de mensagens
    for msg in st.session_state.mensagens:
        with st.chat_message(msg["role"], avatar="ðŸ”¬" if msg["role"] == "user" else "ðŸ¤–"):
            st.markdown(msg["content"])

    # Input do usuÃ¡rio
    if pergunta := st.chat_input("Digite sua pergunta sobre inversores fotovoltaicos..."):

        # Exibe mensagem do usuÃ¡rio
        with st.chat_message("user", avatar="ðŸ”¬"):
            st.markdown(pergunta)

        # Prepara o prompt
        from src.conhecimento.agente import preparar_prompt

        historico_para_prompt = [
            {"role": m["role"], "content": m["content"]}
            for m in st.session_state.mensagens
        ]

        with st.spinner("ðŸ” Buscando na literatura..."):
            prompt, citacoes = preparar_prompt(
                pergunta        = pergunta,
                perfil          = perfil,
                modelo_embeddings = modelo,
                colecao         = colecao,
                historico       = historico_para_prompt,
                colecao_sessoes = colecao_sessoes
            )

        # Gera e exibe resposta com streaming
        with st.chat_message("assistant", avatar="ðŸ¤–"):
            try:
                resposta_texto = st.write_stream(
                    stream_resposta(prompt, st.session_state.llm)
                )

                # Exibe fontes
                if citacoes:
                    st.divider()
                    st.caption("ðŸ“š **Fontes consultadas:**")
                    for arquivo, citacao in citacoes.items():
                        st.caption(f"â†’ {citacao}")

            except Exception as e:
                erro = str(e)
                if "429" in erro:
                    st.error("â³ Limite da API atingido. Troque o provedor no painel lateral ou aguarde.")
                else:
                    st.error(f"âŒ Erro: {e}")
                resposta_texto = f"[Erro: {e}]"

        # Salva no histÃ³rico
        resposta_completa = resposta_texto
        if citacoes:
            resposta_completa += "\n\n**Fontes:** " + ", ".join(citacoes.values())

        st.session_state.mensagens.append({"role": "user",      "content": pergunta})
        st.session_state.mensagens.append({"role": "assistant",  "content": resposta_completa})

        # Auto-salvamento apÃ³s cada interaÃ§Ã£o
        n_interacoes = len(st.session_state.mensagens) // 2
        salvar_sessao_streamlit(pergunta, resposta_completa, n_interacoes, modelo)
