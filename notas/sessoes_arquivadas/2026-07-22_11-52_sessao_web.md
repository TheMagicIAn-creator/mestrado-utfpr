### 1. Código em Python

Em Python, a abordagem padrão e mais limpa utiliza `numpy` para ordenação/vetorização e `matplotlib` para visualização.

```python
import numpy as np
import matplotlib.pyplot as plt

# 1. Gerar dados sintéticos de exemplo (ex.: distribuição Normal)
np.random.seed(42)
dados = np.random.normal(loc=10, scale=2, size=1000)

# 2. Calcular a FDA Empírica (ECDF)
dados_ordenados = np.sort(dados)
p_acumulada = np.arange(1, len(dados_ordenados) + 1) / len(dados_ordenados)

# 3. Plotar o gráfico
plt.figure(figsize=(8, 5))
plt.step(dados_ordenados, p_acumulada, where='post', color='#1f77b4', linewidth=2, label='FDA Empírica')

plt.title('Função de Distribuição Acumulada (FDA / CDF)', fontsize=12)
plt.xlabel('Valor (X)', fontsize=10)
plt.ylabel('Probabilidade Acumulada $P(X \le x)$', fontsize=10)
plt.grid(True, linestyle='--', alpha=0.6)
plt.legend(loc='lower right')
plt.tight_layout()

# Exibir gráfico
plt.show()
```

### 2. Código em C++

Como C++ não possui biblioteca gráfica nativa, o padrão técnico para gráficos é gerar os dados ordenados e renderizar chamando o **Gnuplot** (via *pipe* de sistema) ou salvando um arquivo `.dat` para leitura externa.

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <random>
#include <fstream>
#include <cstdio>

int main() {
    // 1. Gerar dados sintéticos de exemplo (distribuição Normal)
    std::mt19937 gen(42); // Semente fixa para reproducibilidade
    std::normal_distribution<double> dist(10.0, 2.0);

    const int N = 1000;
    std::vector<double> dados(N);
    for (int i = 0; i < N; ++i) {
        dados[i] = dist(gen);
    }

    // 2. Ordenar os dados para cálculo da FDA empírica
    std::sort(dados.begin(), dados.end());

    // 3. Gravar pares (x, P(X <= x)) em arquivo .dat
    std::ofstream file("fda_dados.dat");
    if (!file.is_open()) {
        std::cerr << "Erro ao criar o arquivo de dados." << std::endl;
        return 1;
    }

    for (int i = 0; i < N; ++i) {
        double p_acumulada = static_cast<double>(i + 1) / N;
        file << dados[i] << " " << p_acumulada << "\n";
    }
    file.close();

    // 4. Plotar diretamente via Gnuplot (se instalado no sistema)
    FILE* gnuplotPipe = popen("gnuplot -persistent", "w");
    if (gnuplotPipe) {
        fprintf(gnuplotPipe, "set title 'Função de Distribuição Acumulada (FDA / CDF)'\n");
        fprintf(gnuplotPipe, "set xlabel 'Valor (X)'\n");
        fprintf(gnuplotPipe, "set ylabel 'Probabilidade Acumulada P(X <= x)'\n");
        fprintf(gnuplotPipe, "set grid\n");
        fprintf(gnuplotPipe, "plot 'fda_dados.dat' with steps linecolor rgb 'blue' title 'FDA Empírica'\n");
        fflush(gnuplotPipe);
        pclose(gnuplotPipe);
        std::cout << "Gráfico exibido via Gnuplot e dados salvos em 'fda_dados.dat'." << std::endl;
    } else {
        std::cout << "Dados salvos em 'fda_dados.dat'. (Gnuplot não detectado para exibição automática)." << std::endl;
    }

    return 0;
}
```

O erro no Tkinter ocorre porque o motor dinâmico de layout do Matplotlib (`tight_layout` ou `constrained_layout`) entra em conflito com o cálculo de *bounding box* do renderizador gráfico no PyCharm/Windows quando há janelas interativas e formatação de texto no eixo.

A solução definitiva é **remover o motor de layout automático**, **substituir o caractere MathText do LaTeX por texto simples** e **definir as margens da figura de forma explícita** com `plt.subplots_adjust`.

Aqui está o código corrigido e blindado contra falhas do backend GUI:

```python
import numpy as np
import matplotlib.pyplot as plt

np.random.seed(42)
dados = np.random.normal(loc=10, scale=2, size=1000)

# 2. Calcular a FDA Empírica (ECDF)
dados_ordenados = np.sort(dados)
p_acumulada = np.arange(1, len(dados_ordenados) + 1) / len(dados_ordenados)

# 3. Criar figura e eixos
fig, ax = plt.subplots(figsize=(8, 5))

# 4. Plotar o gráfico
ax.step(dados_ordenados, p_acumulada, where='post', color='#1f77b4', linewidth=2, label='FDA Empírica')

ax.set_title('Função de Distribuição Acumulada (FDA / CDF)', fontsize=12)
ax.set_xlabel('Valor (X)', fontsize=10)
```
