### Script C++ (com geração de dados e chamada do Gnuplot)

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <random>
#include <fstream>

int main() {
    // Gerar dados normais
    std::mt19937 gen(42);
    std::normal_distribution<double> dist(0.0, 1.0);
    
    int n = 100;
    std::vector<double> dados(n);
    for (int i = 0; i < n; ++i) {
        dados[i] = dist(gen);
    }

    // Ordenar dados
    std::sort(dados.begin(), dados.end());

    // Salvar pontos da FDA em arquivo .dat
    std::ofstream file("fda_data.dat");
    for (int i = 0; i < n; ++i) {
        double fda = static_cast<double>(i + 1) / n;
        file << dados[i] << " " << fda << "\n";
    }
    file.close();

    // Plotar via Gnuplot
    FILE* gnuplotPipe = popen("gnuplot -persistent", "w");
    if (gnuplotPipe) {
        fprintf(gnuplotPipe, "set title 'Função de Distribuição Acumulada (FDA / CDF)'\n");
        fprintf(gnuplotPipe, "set xlabel 'Valor'\n");
        fprintf(gnuplotPipe, "set ylabel 'F(x)'\n");
        fprintf(gnuplotPipe, "set grid\n");
        fprintf(gnuplotPipe, "plot 'fda_data.dat' with steps title 'FDA Empírica'\n");
        pclose(gnuplotPipe);
    }

    return 0;
}
```

import numpy as np
import matplotlib.pyplot as plt

def plotar_fda_erros(erros, limiar_p99=None):
    # Ordenação dos dados para FDA empírica
    erros_ordenados = np.sort(erros)
    n = len(erros_ordenados)
    fda_empirica = np.arange(1, n + 1) / n

    # Configuração da figura sem layout automático instável
    fig, ax = plt.subplots(figsize=(8, 5))

    # Curva da FDA Empírica
    ax.step(erros_ordenados, fda_empirica, where="post", label="FDA Empirica", color="navy", linewidth=1.8)

    # Marcação do limiar p99 (se fornecido)
    if limiar_p99 is not None:
        ax.axvline(x=limiar_p99, color="crimson", linestyle="--", linewidth=1.5, label=f"Limiar p99 ({limiar_p99:.4f})")

## Interação 6

**Você:** o código que você mandou era:

import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm

# 1. Definir parâmetros da distribuição (ex.: Média = 10, Desvio Padrão = 2)
media = 10.0
desvio_padrao = 2.0

# 2. Gerar domínio contínuo e denso (1000 pontos suaves entre -4s e +4s)
x = np.linspace(media - 4 * desvio_padrao, media + 4 * desvio_padrao, 1000)

# 3. Calcular a FDA teórica contínua
p_acumulada = norm.cdf(x, loc=media, scale=desvio_padrao)

# 4. Criar figura e eixos
fig, ax = plt.subplots(figsize=(8, 5))

# 5. Plotar como linha contínua (plt.plot em vez de plt.step)
ax.plot(x, p_acumulada, color='#1f77b4', linewidth=2.5, label='FDA Teórica (Normal)')

ax.set_title('Função de Distribuição Acumulada Suave (FDA / CDF)', fontsize=12)
ax.set_xlabel('Valor (X)', fontsize=10)
ax.set_ylabel('Probabilidade Acumulada P(X <= x)', fontsize=10)
ax.grid(True, linestyle='--', alpha=0.6)
ax.legend(loc='lower right')

# 6. Margens explícitas estáticas
plt.subplots_adjust(left=0.12, right=0.95, top=0.90, bottom=0.12)

# 7. Exibir gráfico
plt.show()
